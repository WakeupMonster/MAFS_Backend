class SplashScreen extends StatefulWidget {
  @override
  _SplashScreenState createState() => _SplashScreenState();
}

class _SplashScreenState extends State<SplashScreen> {

  @override
  void initState() {
    super.initState();
    checkLoginAndProfileStatus();
  }

  Future<void> checkLoginAndProfileStatus() async {
    final prefs = await SharedPreferences.getInstance();
    final token = prefs.getString('accessToken');

    if (token != null && token.isNotEmpty) {
      // ✅ Call your API to check profile status
      final res = await api.get('/profile/status');  // Make sure `api` is initialized
      final data = res.data['data'];

      bool canAccessSwipe = data['canAccessSwipe'];
      String nextStepScreen = data['nextStep']['screen'];
      double completionPercentage = data['completionPercentage'];

      if (canAccessSwipe) {
        // ✅ Navigate to Home/Swipe screen
        Navigator.pushReplacementNamed(context, '/home');
      } else {
        // ✅ Resume onboarding from where left
        Navigator.pushReplacementNamed(context, '/$nextStepScreen');

        // ✅ Show progress dialog or message
        showProgress(completionPercentage);
      }
    } else {
      // ✅ Not logged in → Go to Welcome screen
      Navigator.pushReplacementNamed(context, '/welcome');
    }
  }

  void showProgress(double progress) {
    showDialog(
      context: context,
      builder: (_) => AlertDialog(
        title: Text("Profile Incomplete"),
        content: Text("You're ${(progress * 100).toStringAsFixed(0)}% complete!"),
      ),
    );
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      body: Center(child: CircularProgressIndicator()),
    );
  }
}

  const [otpSent, setOtpSent] = useState(false);

    try {
      const response = await api.post('/api/auth/send-otp', {
        phoneNumber: phone,
        countryCode: '+91'
      });

      setOtpSent(true);
      setIsNewUser(response.data.isNewUser);
      
      // Show appropriate message
      if (response.data.isNewUser) {
        showToast('Welcome! Please verify your phone to get started');
      } else {
        showToast('Welcome back! OTP sent to your phone');
      }
    } catch (error) {
      showToast(error.message);
    }
  };

  // Step 2: Verify OTP
  const handleVerifyOTP = async () => {
    try {
      const response = await api.post('/api/auth/verify-otp', {
        phoneNumber: phone,
        countryCode: '+91',
        otp: otp
      });

      // Store tokens
      await AsyncStorage.setItem('accessToken', response.data.accessToken);
      await AsyncStorage.setItem('refreshToken', response.data.refreshToken);
      
      // Store user data in Redux/Context
      dispatch(setUser(response.data.user));

      // Navigate based on nextScreen
      const { nextScreen } = response.data.user;
      
      switch(nextScreen) {

          navigation.reset({

            routes: [{ name: 'HomeTab' }],

          break;
        case 'email_verification':

          navigation.navigate('KYCRejected');

      }
      
    } catch (error) {
